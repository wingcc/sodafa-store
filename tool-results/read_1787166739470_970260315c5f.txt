   287→  function fetchWithTimeout(url) {
   288→    return new Promise(function (resolve, reject) {
   289→      var ctrl = null;
   290→      if (window.AbortController) ctrl = new AbortController();
   291→      var timer = setTimeout(function () {
   292→        if (ctrl) ctrl.abort();
   293→        reject(new Error("timeout"));
   294→      }, FETCH_TIMEOUT);
   295→      fetch(url, ctrl ? { signal: ctrl.signal } : {})
   296→        .then(function (res) {
   297→          if (!res.ok) throw new Error("HTTP " + res.status);
   298→          return res.text();
   299→        })
   300→        .then(function (txt) { clearTimeout(timer);
   301→          resolve(txt); })
   302→        .catch(function (err) { clearTimeout(timer);
   303→          reject(err); });
   304→    });
   305→  }

   311→  function fetchSection(s, attempt) {
   312→    attempt = attempt || 0;
   313→    if (sectionCache[s.id]) return Promise.resolve(sectionCache[s.id]);
   314→    return fetchWithTimeout(cacheBustUrl(s.file))
   315→      .then(function (txt) {
   316→        sectionCache[s.id] = txt;
   317→        cacheSet(s.id, txt);
   318→        return txt;
   319→      })
   320→      .catch(function (err) {
   321→        var cached = cacheGet(s.id);
   322→        if (cached) { sectionCache[s.id] = cached; return cached; }
   323→        if (attempt < FETCH_RETRIES) {
   324→          return new Promise(function (resolve, reject) {
   325→            setTimeout(function () {
   326→              fetchSection(s, attempt + 1).then(resolve, reject);
   327→            }, 400 * (attempt + 1));
   328→          });
   329→        }
   330→        err = err || new Error("unknown");
   331→        err.section = s.id;
   332→        throw err;
   333→      });
   334→  }

   439→  function revealIn(root) {
   440→    if (!rvObs) {
   441→      rvObs = new IntersectionObserver(function (entries) {
   442→        entries.forEach(function (en) {
   443→          if (!en.isIntersecting) return;
   444→          var el = en.target;
   445→          var d = parseInt(el.getAttribute("data-d") || "0", 10);
   446→          setTimeout(function () { el.classList.add("in"); }, d);
   447→          rvObs.unobserve(el);
   448→        });
   449→      }, { threshold: .12 });
   450→    }
   451→    qsa(".rv", root).forEach(function (el) { rvObs.observe(el); });
   452→  }

   455→  function initCounters(root) {
   456→    if (countersObs) countersObs.disconnect();
   457→    var fmt = function (n) { return n.toLocaleString("en-US"); };
   458→    countersObs = new IntersectionObserver(function (entries) {
   459→      entries.forEach(function (en) {
   460→        if (!en.isIntersecting) return;
   461→        var el = en.target;
   462→        var target = parseInt(el.getAttribute("data-count"), 10);
   463→        var pre = el.getAttribute("data-pre") || "";
   464→        var suf = el.getAttribute("data-suf") || "";
   465→        var t0 = null,
   466→          dur = 1600;
   467→        var step = function (ts) {
   468→          if (!t0) t0 = ts;
   469→          var p = Math.min((ts - t0) / dur, 1);
   470→          var e = 1 - Math.pow(1 - p, 3);
   471→          el.textContent = pre + fmt(Math.round(target * e)) + suf;
   472→          if (p < 1) requestAnimationFrame(step);
   473→        };
   474→        requestAnimationFrame(step);
   475→        countersObs.unobserve(el);
   476→      });
   477→    }, { threshold: .6 });
   478→    qsa("[data-count]", root).forEach(function (el) { countersObs.observe(el); });
   479→  }
